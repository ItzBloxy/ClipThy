# ClipThy Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build ClipThy, a lightweight Windows tray app. It keeps the last 15â€“120 s of the primary monitor, system audio and microphone in memory. On Alt+F10 (mixed audio) or Alt+F11 (three audio tracks) it saves them as an MP4 into the ShareX screenshots folder, sorted by month.

**Architecture:**
- A child ffmpeg captures the screen with `ddagrab`, encodes on NVENC, and streams MPEG-TS to stdout. Every frame is stamped with its UTC capture time.
- `VideoBuffer` keeps that stream in RAM, indexed by capture time.
- Two WASAPI captures (through NAudio) keep system audio and mic as timestamped PCM in `AudioBuffer`s.
- On a hotkey, `ClipExporter` cuts all three to the same time span, and a short-lived ffmpeg muxes them into an MP4.
- A WinForms `ApplicationContext` provides the tray icon and wires everything together.

**Tech Stack:** .NET 10 (`net10.0-windows`, WinForms), NAudio.Wasapi 2.2.1, xUnit 2.9.3, ffmpeg 8.0.1 (gyan.dev full build, on PATH).

**Spec:** `docs/superpowers/specs/2026-09-30-clipthy-design.md`. Read Â§5 (architecture) before starting.

```csharp
using System.Diagnostics;

namespace ClipThy.Timing;

/// <summary>Source of "now": QPC ticks for the app's timeline, plus UTC for file names.</summary>
public interface IClock
{
    long QpcNow { get; }
    long Frequency { get; }
    DateTime UtcNow { get; }
}

public sealed class SystemClock : IClock
{
    public static readonly SystemClock Instance = new();

    public long QpcNow => Stopwatch.GetTimestamp();
    public long Frequency => Stopwatch.Frequency;
    public DateTime UtcNow => DateTime.UtcNow;
}

/// <summary>Conversions between QPC ticks and other units, written to avoid 64-bit overflow.</summary>
public static class QpcTime
{
    private const long HundredNsPerSecond = 10_000_000;

    public static long FromSeconds(double seconds, long frequency) => (long)Math.Round(seconds * frequency);

    public static long FromTimeSpan(TimeSpan span, long frequency) => FromSeconds(span.TotalSeconds, frequency);

    public static double ToSeconds(long ticks, long frequency) => (double)ticks / frequency;

    /// <summary>WASAPI reports capture times as QPC positions in 100-ns units.</summary>
    public static long From100ns(long hundredNs, long frequency) =>
        hundredNs / HundredNsPerSecond * frequency + hundredNs % HundredNsPerSecond * frequency / HundredNsPerSecond;

    /// <summary>Duration of <paramref name="frames"/> audio frames at <paramref name="rate"/> Hz, in ticks.</summary>
    public static long FromFrames(long frames, int rate, long frequency) =>
        frames / rate * frequency + frames % rate * frequency / rate;

    /// <summary>Number of audio frames at <paramref name="rate"/> Hz in <paramref name="ticks"/>, rounded to nearest.</summary>
    public static long ToFrames(long ticks, int rate, long frequency) => (long)Math.Round((double)ticks * rate / frequency);
}
```

```csharp
namespace ClipThy.Hotkeys;

/// <summary>Modifier flags with the values RegisterHotKey expects (MOD_ALT, MOD_CONTROL, MOD_SHIFT, MOD_WIN).</summary>
[Flags]
public enum HotkeyModifiers
{
    None = 0,
    Alt = 0x1,
    Control = 0x2,
    Shift = 0x4,
    Win = 0x8,
}

/// <summary>A global hotkey such as "Alt+F10". <see cref="Key"/> is a Windows virtual-key code.</summary>
public readonly record struct Hotkey(HotkeyModifiers Modifiers, Keys Key)
{
    public static bool TryParse(string? text, out Hotkey hotkey)
    {
        hotkey = default;
        if (string.IsNullOrWhiteSpace(text)) return false;

        string[] parts = text.Split('+', StringSplitOptions.TrimEntries);
        if (parts.Any(p => p.Length == 0)) return false;

        var modifiers = HotkeyModifiers.None;
        foreach (string part in parts[..^1])
        {
            HotkeyModifiers modifier = part.ToLowerInvariant() switch
            {
                "alt" => HotkeyModifiers.Alt,
                "ctrl" or "control" => HotkeyModifiers.Control,
                "shift" => HotkeyModifiers.Shift,
                "win" => HotkeyModifiers.Win,
                _ => HotkeyModifiers.None,
            };
            if (modifier == HotkeyModifiers.None || modifiers.HasFlag(modifier)) return false;
            modifiers |= modifier;
        }

        if (!TryParseKey(parts[^1], out Keys key)) return false;
        hotkey = new Hotkey(modifiers, key);
        return true;
    }

    private static bool TryParseKey(string token, out Keys key)
    {
        key = Keys.None;
        if (token.Length == 1 && char.IsAsciiDigit(token[0]))
        {
            key = Keys.D0 + (token[0] - '0');
            return true;
        }

        // Reject numbers ("10" would parse as a raw enum value) and flag lists ("A, B").
        if (token.All(char.IsAsciiDigit) || token.Contains(',')) return false;
        if (!Enum.TryParse(token, ignoreCase: true, out key)) return false;

        bool isModifierKey = key is Keys.ShiftKey or Keys.ControlKey or Keys.Menu or Keys.LWin or Keys.RWin;
        bool isPlainKeyCode = key != Keys.None && ((int)key & ~0xFF) == 0;
        return isPlainKeyCode && !isModifierKey;
    }

    public override string ToString()
    {
        var parts = new List<string>();
        if (Modifiers.HasFlag(HotkeyModifiers.Control)) parts.Add("Ctrl");
        if (Modifiers.HasFlag(HotkeyModifiers.Alt)) parts.Add("Alt");
        if (Modifiers.HasFlag(HotkeyModifiers.Shift)) parts.Add("Shift");
        if (Modifiers.HasFlag(HotkeyModifiers.Win)) parts.Add("Win");
        parts.Add(Key.ToString());
        return string.Join('+', parts);
    }
}
```

```csharp
public static class WavWriter
{
    public static void Write(string path, ReadOnlySpan<byte> pcm, int sampleRate = 48_000, short channels = 2, short bitsPerSample = 16)
    {
        using FileStream file = File.Create(path);
        Write(file, pcm, sampleRate, channels, bitsPerSample);
    }

    public static void Write(Stream stream, ReadOnlySpan<byte> pcm, int sampleRate, short channels, short bitsPerSample)
    {
        int blockAlign = channels * bitsPerSample / 8;
        Span<byte> header = stackalloc byte[44];
        "RIFF"u8.CopyTo(header);
        BinaryPrimitives.WriteInt32LittleEndian(header[4..], 36 + pcm.Length);
        "WAVE"u8.CopyTo(header[8..]);
        "fmt "u8.CopyTo(header[12..]);
        BinaryPrimitives.WriteInt32LittleEndian(header[16..], 16);
        BinaryPrimitives.WriteInt16LittleEndian(header[20..], 1); // PCM
        BinaryPrimitives.WriteInt16LittleEndian(header[22..], channels);
        BinaryPrimitives.WriteInt32LittleEndian(header[24..], sampleRate);
        BinaryPrimitives.WriteInt32LittleEndian(header[28..], sampleRate * blockAlign);
        BinaryPrimitives.WriteInt16LittleEndian(header[32..], (short)blockAlign);
        BinaryPrimitives.WriteInt16LittleEndian(header[34..], bitsPerSample);
        "data"u8.CopyTo(header[36..]);
        BinaryPrimitives.WriteInt32LittleEndian(header[40..], pcm.Length);
        stream.Write(header);
        stream.Write(pcm);
    }
}
```

```csharp
using System.Buffers.Binary;

namespace ClipThy.Tests.TestSupport;

/// <summary>Builds small, valid MPEG-TS streams (PAT, PMT and H.264 PES packets) for tests.</summary>
public sealed class TsBuilder
{
    public const int PmtPid = 0x1000;
    public const int VideoPid = 0x100;
    public const int PacketsPerFrame = 3;

    private int _patCounter;
    private int _pmtCounter;
    private int _videoCounter;

    public byte[] Pat()
    {
        byte[] section =
        [
            0x00, 0xB0, 0x0D, 0x00, 0x01, 0xC1, 0x00, 0x00, // table header, section_length 13
            0x00, 0x01, 0xE0 | (PmtPid >> 8), PmtPid & 0xFF, // program 1 -> PMT PID
        ];
        return SectionPacket(0x0000, ref _patCounter, section);
    }

    public byte[] Pmt()
    {
        byte[] section =
        [
            0x02, 0xB0, 0x12, 0x00, 0x01, 0xC1, 0x00, 0x00, // table header, section_length 18
            0xE0 | (VideoPid >> 8), VideoPid & 0xFF,        // PCR PID
            0xF0, 0x00,                                     // program_info_length 0
            0x1B, 0xE0 | (VideoPid >> 8), VideoPid & 0xFF, 0xF0, 0x00, // H.264 stream
        ];
        return SectionPacket(PmtPid, ref _pmtCounter, section);
    }

    /// <summary>One video frame; the first packet carries the PES header (and the keyframe flag).</summary>
    public byte[] Frame(long pts, bool keyframe, int payloadPackets = PacketsPerFrame)
    {
        var packets = new byte[payloadPackets * 188];
        for (int i = 0; i < payloadPackets; i++)
        {
            Span<byte> p = packets.AsSpan(i * 188, 188);
            p[0] = 0x47;
            p[1] = (byte)((i == 0 ? 0x40 : 0x00) | (VideoPid >> 8));
            p[2] = VideoPid & 0xFF;
            int offset = 4;
            if (i == 0 && keyframe)
            {
                p[3] = (byte)(0x30 | (_videoCounter++ & 0x0F)); // adaptation field + payload
                p[4] = 1;                                       // adaptation_field_length
                p[5] = 0x40;                                    // random_access_indicator
                offset = 6;
            }
            else
            {
                p[3] = (byte)(0x10 | (_videoCounter++ & 0x0F)); // payload only
            }
            if (i == 0)
            {
                byte[] pes = PesHeader(pts);
                pes.CopyTo(p[offset..]);
                offset += pes.Length;
            }
            p[offset..].Fill(0xAB); // stand-in for H.264 data
        }
        return packets;
    }

    private static byte[] PesHeader(long pts) =>
    [
        0x00, 0x00, 0x01, 0xE0, 0x00, 0x00, // start code, stream id, unbounded length
        0x80, 0x80, 0x05,                   // flags: PTS present, header length 5
        (byte)(0x21 | ((pts >> 29) & 0x0E)),
        (byte)(pts >> 22),
        (byte)(0x01 | ((pts >> 14) & 0xFE)),
        (byte)(pts >> 7),
        (byte)(0x01 | ((pts << 1) & 0xFE)),
    ];

    private static byte[] SectionPacket(int pid, ref int counter, byte[] section)
    {
        var packet = new byte[188];
        packet.AsSpan().Fill(0xFF);
        packet[0] = 0x47;
        packet[1] = (byte)(0x40 | (pid >> 8));
        packet[2] = (byte)(pid & 0xFF);
        packet[3] = (byte)(0x10 | (counter++ & 0x0F));
        packet[4] = 0x00; // pointer_field
        section.CopyTo(packet, 5);
        BinaryPrimitives.WriteUInt32BigEndian(packet.AsSpan(5 + section.Length), Crc32Mpeg2(section));
        return packet;
    }

    private static uint Crc32Mpeg2(ReadOnlySpan<byte> data)
    {
        uint crc = 0xFFFFFFFF;
        foreach (byte b in data)
        {
            crc ^= (uint)b << 24;
            for (int bit = 0; bit < 8; bit++) crc = (crc & 0x80000000) != 0 ? (crc << 1) ^ 0x04C11DB7 : crc << 1;
        }
        return crc;
    }
}
```

```csharp
namespace ClipThy.Video;

/// <summary>Minimal MPEG-TS parsing: just what VideoBuffer needs.</summary>
public static class MpegTs
{
    public const int PacketSize = 188;
    public const byte SyncByte = 0x47;
    public const int ClockRate = 90_000;
    public const long PtsWrap = 1L << 33;

    public static int Pid(ReadOnlySpan<byte> packet) => ((packet[1] & 0x1F) << 8) | packet[2];

    public static bool PayloadUnitStart(ReadOnlySpan<byte> packet) => (packet[1] & 0x40) != 0;

    /// <summary>True when the adaptation field's random_access_indicator is set (ffmpeg sets it on keyframes).</summary>
    public static bool RandomAccess(ReadOnlySpan<byte> packet) =>
        (packet[3] & 0x20) != 0 && packet[4] > 0 && (packet[5] & 0x40) != 0;

    public static ReadOnlySpan<byte> Payload(ReadOnlySpan<byte> packet)
    {
        int adaptationControl = (packet[3] >> 4) & 0x3;
        if ((adaptationControl & 0x1) == 0) return ReadOnlySpan<byte>.Empty;
        int offset = (adaptationControl & 0x2) != 0 ? 5 + packet[4] : 4;
        return offset < PacketSize ? packet[offset..PacketSize] : ReadOnlySpan<byte>.Empty;
    }

    /// <summary>Reads the PTS of a PES header that starts at the beginning of <paramref name="payload"/>.</summary>
    public static bool TryReadPts(ReadOnlySpan<byte> payload, out long pts)
    {
        pts = 0;
        if (payload.Length < 14 || payload[0] != 0 || payload[1] != 0 || payload[2] != 1) return false;
        if ((payload[7] & 0x80) == 0) return false;
        ReadOnlySpan<byte> b = payload.Slice(9, 5);
        pts = ((long)((b[0] >> 1) & 0x07) << 30) | ((long)b[1] << 22) | ((long)(b[2] >> 1) << 15)
            | ((long)b[3] << 7) | ((long)b[4] >> 1);
        return true;
    }

    /// <summary>Reads a PAT and returns the PID of the first program's PMT.</summary>
    public static bool TryReadPmtPid(ReadOnlySpan<byte> payload, out int pmtPid)
    {
        pmtPid = 0;
        if (!TryGetSection(payload, tableId: 0x00, out ReadOnlySpan<byte> section)) return false;
        for (int i = 8; i + 4 <= section.Length - 4; i += 4)
        {
            int programNumber = (section[i] << 8) | section[i + 1];
            if (programNumber == 0) continue; // network information PID
            pmtPid = ((section[i + 2] & 0x1F) << 8) | section[i + 3];
            return true;
        }
        return false;
    }

    /// <summary>Reads a PMT and returns the PID of the first H.264 stream.</summary>
    public static bool TryReadH264Pid(ReadOnlySpan<byte> payload, out int videoPid)
    {
        videoPid = 0;
        if (!TryGetSection(payload, tableId: 0x02, out ReadOnlySpan<byte> section) || section.Length < 16) return false;
        int programInfoLength = ((section[10] & 0x0F) << 8) | section[11];
        for (int i = 12 + programInfoLength; i + 5 <= section.Length - 4;)
        {
            int streamType = section[i];
            int pid = ((section[i + 1] & 0x1F) << 8) | section[i + 2];
            int esInfoLength = ((section[i + 3] & 0x0F) << 8) | section[i + 4];
            if (streamType == 0x1B)
            {
                videoPid = pid;
                return true;
            }
            i += 5 + esInfoLength;
        }
        return false;
    }

    /// <summary>UTC time in the 90 kHz, 33-bit clock that ffmpeg writes as PTS with -use_wallclock_as_timestamps.</summary>
    public static long UtcTo90k(DateTime utc) => (utc.Ticks - DateTime.UnixEpoch.Ticks) * 9 / 1000 & (PtsWrap - 1);

    private static bool TryGetSection(ReadOnlySpan<byte> payload, byte tableId, out ReadOnlySpan<byte> section)
    {
        section = default;
        if (payload.IsEmpty) return false;
        int start = 1 + payload[0]; // skip pointer_field
        if (start + 3 > payload.Length || payload[start] != tableId) return false;
        int sectionLength = ((payload[start + 1] & 0x0F) << 8) | payload[start + 2];
        int end = start + 3 + sectionLength;
        if (sectionLength < 9 || end > payload.Length) return false;
        section = payload[start..end];
        return true;
    }
}
```

### Task 8: Video buffer

This is the core of the design (spec Â§5.1â€“5.2).

**Converting a frame's PTS to a capture time.** When a frame's first packet arrives, its PTS (UTC capture time) is converted to QPC:
- `delay = (arrivalUtc90k âˆ’ pts) mod 2Â³Â³`, made signed;
- `captureQpc = arrivalQpc âˆ’ delay`.

Delays from âˆ’1 s to 5 s are valid. Small negative delays are normal: ffmpeg's clock can run a few ms ahead of .NET's, and NVENC often delivers a frame inside that margin. An integration test caught this with a 0-based window. Anything outside âˆ’1â€¦5 s falls back to the arrival time and increments `TimestampWarnings`.

**Storage.** Packets are stored in 770 KB blocks, and up to 4 spare blocks are reused, so there are no per-packet allocations.

**Snapshot.** A snapshot contains PAT + PMT, then:
- it starts at the last keyframe at or before `end âˆ’ length`;
- it stops before the first frame captured after `end`.

When no frame was captured after `end`, the newest frame is left out, because it may still be arriving.

**Tests.** The tests use a QPC frequency of 90 kHz, so expected values are exact integers.

```csharp
using ClipThy.Timing;

namespace ClipThy.Video;

/// <summary>A self-contained piece of the stream: PAT + PMT + whole frames starting at a keyframe.</summary>
public sealed record VideoSnapshot(byte[] Data, long FirstFrameQpc, long LastFrameQpc, int FrameCount);

public interface IVideoSource
{
    /// <summary>Capture time of the newest frame that has fully arrived, or null if there is none.</summary>
    long? NewestCaptureQpc { get; }

    VideoSnapshot? Snapshot(long endQpc, TimeSpan length);
}

/// <summary>
/// Keeps the capture ffmpeg's recent MPEG-TS output in memory, indexed by each frame's capture time.
/// Every frame's PTS is its UTC capture time (-use_wallclock_as_timestamps), converted to QPC on arrival.
/// </summary>
public sealed class VideoBuffer : IVideoSource
{
    private const int PacketsPerBlock = 4096;
    private const int MaxSpareBlocks = 4;
    private const long MinDelay90k = -MpegTs.ClockRate; // ffmpeg's clock can run a few ms ahead of .NET's
    private const long MaxDelay90k = 5L * MpegTs.ClockRate;

    private readonly object _lock = new();
    private readonly long _frequency;
    private readonly List<byte[]> _blocks = [];
    private readonly Stack<byte[]> _spareBlocks = new();
    private readonly List<FrameEntry> _frames = [];
    private readonly byte[] _partial = new byte[MpegTs.PacketSize];
    private int _partialLength;
    private long _firstBlockNumber;
    private long _packetCount;
    private byte[]? _pat;
    private byte[]? _pmt;
    private int _pmtPid = -1;
    private int _videoPid = -1;
    private long _timestampWarnings;
    private TimeSpan _retention = TimeSpan.FromSeconds(32);

    private readonly record struct FrameEntry(long PacketIndex, long CaptureQpc, bool Keyframe);

    public VideoBuffer(long qpcFrequency) => _frequency = qpcFrequency;

    /// <summary>How much capture time to keep. Set to clip length + 2 s.</summary>
    public TimeSpan Retention
    {
        get { lock (_lock) return _retention; }
        set { lock (_lock) _retention = value; }
    }

    /// <summary>Frames whose PTS couldn't be mapped (delay outside −1…5 s); they use their arrival time instead.</summary>
    public long TimestampWarnings
    {
        get { lock (_lock) return _timestampWarnings; }
    }

    public long? NewestCaptureQpc
    {
        get { lock (_lock) return _frames.Count >= 2 ? _frames[^2].CaptureQpc : null; }
    }

    /// <summary>Adds bytes read from ffmpeg's stdout. <paramref name="data"/> doesn't need to be packet-aligned.</summary>
    public void Append(ReadOnlySpan<byte> data, long arrivalQpc, long arrivalUtc90k)
    {
        lock (_lock)
        {
            while (!data.IsEmpty)
            {
                if (_partialLength > 0 || data.Length < MpegTs.PacketSize)
                {
                    int take = Math.Min(MpegTs.PacketSize - _partialLength, data.Length);
                    data[..take].CopyTo(_partial.AsSpan(_partialLength));
                    _partialLength += take;
                    data = data[take..];
                    if (_partialLength == MpegTs.PacketSize)
                    {
                        AddPacket(_partial, arrivalQpc, arrivalUtc90k);
                        _partialLength = 0;
                    }
                }
                else
                {
                    AddPacket(data[..MpegTs.PacketSize], arrivalQpc, arrivalUtc90k);
                    data = data[MpegTs.PacketSize..];
                }
            }
        }
    }

    public VideoSnapshot? Snapshot(long endQpc, TimeSpan length)
    {
        lock (_lock)
        {
            if (_pat is null || _pmt is null) return null;

            long startTarget = endQpc - QpcTime.FromTimeSpan(length, _frequency);
            int start = -1;
            int oldestKeyframe = -1;
            for (int i = 0; i < _frames.Count; i++)
            {
                if (!_frames[i].Keyframe) continue;
                if (oldestKeyframe < 0) oldestKeyframe = i;
                if (_frames[i].CaptureQpc > startTarget) break;
                start = i;
            }
            if (start < 0) start = oldestKeyframe;
            if (start < 0) return null;

            // Stop before the first frame captured after endQpc. If there is none, leave out the newest
            // frame because its packets may still be arriving.
            int end = _frames.Count - 1;
            for (int i = start; i < _frames.Count; i++)
            {
                if (_frames[i].CaptureQpc > endQpc)
                {
                    end = i;
                    break;
                }
            }
            if (end <= start) return null;

            long fromPacket = _frames[start].PacketIndex;
            long toPacket = _frames[end].PacketIndex;
            var data = new byte[(2 + toPacket - fromPacket) * MpegTs.PacketSize];
            _pat.CopyTo(data, 0);
            _pmt.CopyTo(data, MpegTs.PacketSize);
            CopyPackets(fromPacket, toPacket, data.AsSpan(2 * MpegTs.PacketSize));
            return new VideoSnapshot(data, _frames[start].CaptureQpc, _frames[end - 1].CaptureQpc, end - start);
        }
    }

    /// <summary>Forgets everything; called whenever ffmpeg restarts (a new stream begins).</summary>
    public void Clear()
    {
        lock (_lock)
        {
            foreach (byte[] block in _blocks) ReturnBlock(block);
            _blocks.Clear();
            _frames.Clear();
            _partialLength = 0;
            _firstBlockNumber = 0;
            _packetCount = 0;
            _pat = null;
            _pmt = null;
            _pmtPid = -1;
            _videoPid = -1;
        }
    }

    private void AddPacket(ReadOnlySpan<byte> packet, long arrivalQpc, long arrivalUtc90k)
    {
        if (packet[0] != MpegTs.SyncByte) return;

        int pid = MpegTs.Pid(packet);
        if (pid == 0 && MpegTs.PayloadUnitStart(packet) && MpegTs.TryReadPmtPid(MpegTs.Payload(packet), out int pmtPid))
        {
            _pat = packet.ToArray();
            _pmtPid = pmtPid;
        }
        else if (pid == _pmtPid && MpegTs.PayloadUnitStart(packet) && MpegTs.TryReadH264Pid(MpegTs.Payload(packet), out int videoPid))
        {
            _pmt = packet.ToArray();
            _videoPid = videoPid;
        }

        StorePacket(packet);

        if (pid == _videoPid && MpegTs.PayloadUnitStart(packet) && MpegTs.TryReadPts(MpegTs.Payload(packet), out long pts))
        {
            bool keyframe = MpegTs.RandomAccess(packet);
            _frames.Add(new FrameEntry(_packetCount, ToCaptureQpc(pts, arrivalQpc, arrivalUtc90k), keyframe));
            if (keyframe) Trim();
        }
        _packetCount++;
    }

    private long ToCaptureQpc(long pts, long arrivalQpc, long arrivalUtc90k)
    {
        long delay90k = ((arrivalUtc90k - pts) % MpegTs.PtsWrap + MpegTs.PtsWrap) % MpegTs.PtsWrap;
        if (delay90k > MpegTs.PtsWrap / 2) delay90k -= MpegTs.PtsWrap; // make it signed
        if (delay90k < MinDelay90k || delay90k > MaxDelay90k)
        {
            _timestampWarnings++;
            return arrivalQpc;
        }
        return arrivalQpc - delay90k * _frequency / MpegTs.ClockRate;
    }

    private void StorePacket(ReadOnlySpan<byte> packet)
    {
        long blockNumber = _packetCount / PacketsPerBlock;
        int indexInBlock = (int)(_packetCount % PacketsPerBlock);
        if (indexInBlock == 0) _blocks.Add(RentBlock());
        packet.CopyTo(_blocks[(int)(blockNumber - _firstBlockNumber)].AsSpan(indexInBlock * MpegTs.PacketSize));
    }

    private void CopyPackets(long fromPacket, long toPacket, Span<byte> destination)
    {
        long packet = fromPacket;
        int written = 0;
        while (packet < toPacket)
        {
            long blockNumber = packet / PacketsPerBlock;
            int indexInBlock = (int)(packet % PacketsPerBlock);
            int count = (int)Math.Min(PacketsPerBlock - indexInBlock, toPacket - packet);
            _blocks[(int)(blockNumber - _firstBlockNumber)]
                .AsSpan(indexInBlock * MpegTs.PacketSize, count * MpegTs.PacketSize)
                .CopyTo(destination[(written * MpegTs.PacketSize)..]);
            written += count;
            packet += count;
        }
    }

    /// <summary>Drops frames older than the retention window, keeping the keyframe at or before its start.</summary>
    private void Trim()
    {
        long keepFrom = _frames[^1].CaptureQpc - QpcTime.FromTimeSpan(_retention, _frequency);
        int cut = -1;
        for (int i = 0; i < _frames.Count && _frames[i].CaptureQpc <= keepFrom; i++)
        {
            if (_frames[i].Keyframe) cut = i;
        }
        if (cut <= 0) return;

        _frames.RemoveRange(0, cut);
        long firstNeededBlock = _frames[0].PacketIndex / PacketsPerBlock;
        while (_firstBlockNumber < firstNeededBlock)
        {
            ReturnBlock(_blocks[0]);
            _blocks.RemoveAt(0);
            _firstBlockNumber++;
        }
    }

    private byte[] RentBlock() =>
        _spareBlocks.Count > 0 ? _spareBlocks.Pop() : new byte[PacketsPerBlock * MpegTs.PacketSize];

    private void ReturnBlock(byte[] block)
    {
        if (_spareBlocks.Count < MaxSpareBlocks) _spareBlocks.Push(block);
    }
}
```

```csharp
using ClipThy.Timing;

namespace ClipThy.Audio;

public interface IAudioSource
{
    /// <summary>Exactly [fromQpc, toQpc) as 48 kHz 16-bit stereo PCM, with silence wherever nothing was captured.</summary>
    byte[] Extract(long fromQpc, long toQpc);
}

/// <summary>
/// Recent audio in memory, as "runs" of back-to-back packets. Each run knows the capture time of its
/// first frame; a packet that doesn't line up with the end of the current run (±15 ms) starts a new run.
/// </summary>
public sealed class AudioBuffer : IAudioSource
{
    public const int SampleRate = 48_000;
    public const int BytesPerFrame = 4; // 16-bit stereo
    private const int ChunkFrames = SampleRate; // 1 s per pooled chunk
    private const int ChunkBytes = ChunkFrames * BytesPerFrame;
    private const int MaxSpareChunks = 4;

    private readonly object _lock = new();
    private readonly long _frequency;
    private readonly long _toleranceTicks;
    private readonly List<Run> _runs = [];
    private readonly Stack<byte[]> _spareChunks = new();
    private TimeSpan _retention = TimeSpan.FromSeconds(33);

    public AudioBuffer(long qpcFrequency)
    {
        _frequency = qpcFrequency;
        _toleranceTicks = QpcTime.FromSeconds(0.015, qpcFrequency);
    }

    /// <summary>How much audio to keep. Set to clip length + 3 s.</summary>
    public TimeSpan Retention
    {
        get { lock (_lock) return _retention; }
        set { lock (_lock) _retention = value; }
    }

    public long? NewestEndQpc
    {
        get { lock (_lock) return _runs.Count == 0 ? null : _runs[^1].EndQpc(_frequency); }
    }

    public void Append(ReadOnlySpan<byte> pcm, long startQpc)
    {
        if (pcm.Length % BytesPerFrame != 0)
        {
            throw new ArgumentException("PCM must be whole 16-bit stereo frames.", nameof(pcm));
        }
        if (pcm.IsEmpty) return;

        lock (_lock)
        {
            Run? run = _runs.Count > 0 ? _runs[^1] : null;
            if (run is null || Math.Abs(startQpc - run.EndQpc(_frequency)) > _toleranceTicks)
            {
                run = new Run(startQpc);
                _runs.Add(run);
            }
            run.Write(pcm, RentChunk);
            Trim();
        }
    }

    public byte[] Extract(long fromQpc, long toQpc)
    {
        long frames = QpcTime.ToFrames(toQpc - fromQpc, SampleRate, _frequency);
        if (frames <= 0) return [];
        var result = new byte[frames * BytesPerFrame]; // zeros = silence

        lock (_lock)
        {
            // Runs are in capture order, so where two runs overlap the later one wins.
            foreach (Run run in _runs)
            {
                long from = Math.Max(fromQpc, run.TimeOf(run.FirstRetainedFrame, _frequency));
                long to = Math.Min(toQpc, run.EndQpc(_frequency));
                if (from >= to) continue;

                long sourceFrame = Math.Max(run.FirstRetainedFrame, QpcTime.ToFrames(from - run.StartQpc, SampleRate, _frequency));
                long targetFrame = QpcTime.ToFrames(from - fromQpc, SampleRate, _frequency);
                long count = Math.Min(run.FrameCount - sourceFrame, frames - targetFrame);
                if (count <= 0) continue;

                run.CopyFrames(sourceFrame, count, result.AsSpan((int)(targetFrame * BytesPerFrame), (int)(count * BytesPerFrame)));
            }
        }
        return result;
    }

    private void Trim()
    {
        long keepFrom = _runs[^1].EndQpc(_frequency) - QpcTime.FromTimeSpan(_retention, _frequency);
        while (_runs.Count > 0)
        {
            Run oldest = _runs[0];
            if (oldest.EndQpc(_frequency) <= keepFrom)
            {
                foreach (byte[] chunk in oldest.Chunks) ReturnChunk(chunk);
                _runs.RemoveAt(0);
                continue;
            }
            while (oldest.Chunks.Count > 1 && oldest.TimeOf((oldest.FirstChunk + 1) * ChunkFrames, _frequency) <= keepFrom)
            {
                ReturnChunk(oldest.DropFirstChunk());
            }
            break;
        }
    }

    private byte[] RentChunk() => _spareChunks.Count > 0 ? _spareChunks.Pop() : new byte[ChunkBytes];

    private void ReturnChunk(byte[] chunk)
    {
        if (_spareChunks.Count < MaxSpareChunks) _spareChunks.Push(chunk);
    }

    private sealed class Run(long startQpc)
    {
        public long StartQpc { get; } = startQpc;

        /// <summary>Frames written since the run started (frame indices 0..FrameCount-1).</summary>
        public long FrameCount { get; private set; }

        /// <summary>Chunk index of Chunks[0]; earlier chunks have been trimmed.</summary>
        public long FirstChunk { get; private set; }

        public List<byte[]> Chunks { get; } = [];

        public long FirstRetainedFrame => FirstChunk * ChunkFrames;

        public long TimeOf(long frame, long frequency) => StartQpc + QpcTime.FromFrames(frame, SampleRate, frequency);

        public long EndQpc(long frequency) => TimeOf(FrameCount, frequency);

        public void Write(ReadOnlySpan<byte> pcm, Func<byte[]> rentChunk)
        {
            while (!pcm.IsEmpty)
            {
                long chunk = FrameCount / ChunkFrames;
                int frameInChunk = (int)(FrameCount % ChunkFrames);
                if (chunk - FirstChunk == Chunks.Count) Chunks.Add(rentChunk());
                int frames = Math.Min(ChunkFrames - frameInChunk, pcm.Length / BytesPerFrame);
                pcm[..(frames * BytesPerFrame)].CopyTo(Chunks[(int)(chunk - FirstChunk)].AsSpan(frameInChunk * BytesPerFrame));
                pcm = pcm[(frames * BytesPerFrame)..];
                FrameCount += frames;
            }
        }

        public void CopyFrames(long fromFrame, long count, Span<byte> destination)
        {
            int written = 0;
            while (count > 0)
            {
                long chunk = fromFrame / ChunkFrames;
                int frameInChunk = (int)(fromFrame % ChunkFrames);
                int frames = (int)Math.Min(ChunkFrames - frameInChunk, count);
                Chunks[(int)(chunk - FirstChunk)]
                    .AsSpan(frameInChunk * BytesPerFrame, frames * BytesPerFrame)
                    .CopyTo(destination[(written * BytesPerFrame)..]);
                written += frames;
                fromFrame += frames;
                count -= frames;
            }
        }

        public byte[] DropFirstChunk()
        {
            byte[] chunk = Chunks[0];
            Chunks.RemoveAt(0);
            FirstChunk++;
            return chunk;
        }
    }
}
```

### Task 10: PCM converter

- mono is duplicated to both channels;

```csharp
using System.Runtime.InteropServices;

namespace ClipThy.Audio;

public enum SampleFormat
{
    Float32,
    Pcm16,
}

/// <summary>A WASAPI mix format that ClipThy can convert.</summary>
public sealed record InputFormat(int SampleRate, int Channels, SampleFormat Format)
{
    public int BytesPerFrame => Channels * (Format == SampleFormat.Float32 ? 4 : 2);
}

/// <summary>
/// Converts captured audio to 48 kHz 16-bit stereo. Streaming: resampler state carries over between calls,
/// so one converter must be used per continuous capture stream.
/// </summary>
public sealed class PcmConverter
{
    private const float Minus3dB = 0.70710677f;

    private readonly InputFormat _input;
    private readonly double _step; // input frames per output frame
    private double _position;      // next output position in input frames; -1 = last frame of the previous call
    private float _previousLeft;
    private float _previousRight;
    private float[] _interleaved = new float[4096];
    private float[] _stereo = new float[4096];
    private byte[] _output = new byte[8192];

    public PcmConverter(InputFormat input)
    {
        if (input.Channels < 1 || input.SampleRate < 8000)
        {
            throw new ArgumentOutOfRangeException(nameof(input), input, "Unsupported audio format.");
        }
        _input = input;
        _step = (double)input.SampleRate / AudioBuffer.SampleRate;
    }

    /// <summary>Returns 48 kHz stereo PCM16 for <paramref name="input"/>; valid until the next call.</summary>
    public ReadOnlySpan<byte> Convert(ReadOnlySpan<byte> input)
    {
        int frames = input.Length / _input.BytesPerFrame;
        if (frames == 0) return ReadOnlySpan<byte>.Empty;

        Grow(ref _interleaved, frames * _input.Channels);
        Grow(ref _stereo, frames * 2);
        Decode(input[..(frames * _input.BytesPerFrame)]);
        Downmix(frames);
        return _input.SampleRate == AudioBuffer.SampleRate ? Quantize(frames) : Resample(frames);
    }

    private void Decode(ReadOnlySpan<byte> input)
    {
        if (_input.Format == SampleFormat.Float32)
        {
            MemoryMarshal.Cast<byte, float>(input).CopyTo(_interleaved);
            return;
        }
        ReadOnlySpan<short> samples = MemoryMarshal.Cast<byte, short>(input);
        for (int i = 0; i < samples.Length; i++) _interleaved[i] = samples[i] / 32768f;
    }

    /// <summary>Mono is duplicated; 3+ channels use the standard order FL FR FC LFE then surround pairs.</summary>
    private void Downmix(int frames)
    {
        int channels = _input.Channels;
        for (int i = 0; i < frames; i++)
        {
            int f = i * channels;
            float left, right;
            if (channels == 1)
            {
                left = right = _interleaved[f];
            }
            else
            {
                left = _interleaved[f];
                right = _interleaved[f + 1];
                if (channels >= 3)
                {
                    left += Minus3dB * _interleaved[f + 2];
                    right += Minus3dB * _interleaved[f + 2];
                }
                for (int c = 4; c < channels; c++) // channel 3 is LFE and is dropped
                {
                    if (c % 2 == 0) left += Minus3dB * _interleaved[f + c];
                    else right += Minus3dB * _interleaved[f + c];
                }
            }
            _stereo[2 * i] = left;
            _stereo[2 * i + 1] = right;
        }
    }

    private ReadOnlySpan<byte> Quantize(int frames)
    {
        Grow(ref _output, frames * AudioBuffer.BytesPerFrame);
        Span<short> output = MemoryMarshal.Cast<byte, short>(_output.AsSpan());
        for (int i = 0; i < frames * 2; i++) output[i] = ToPcm16(_stereo[i]);
        return _output.AsSpan(0, frames * AudioBuffer.BytesPerFrame);
    }

    /// <summary>Linear interpolation; frame -1 is the last frame of the previous call.</summary>
    private ReadOnlySpan<byte> Resample(int frames)
    {
        int maxOutput = (int)Math.Ceiling((frames + 1) / _step) + 2;
        Grow(ref _output, maxOutput * AudioBuffer.BytesPerFrame);
        Span<short> output = MemoryMarshal.Cast<byte, short>(_output.AsSpan());

        int produced = 0;
        while (_position < frames - 1)
        {
            int i = (int)Math.Floor(_position);
            float t = (float)(_position - i);
            float left0 = i < 0 ? _previousLeft : _stereo[2 * i];
            float right0 = i < 0 ? _previousRight : _stereo[2 * i + 1];
            float left1 = _stereo[2 * (i + 1)];
            float right1 = _stereo[2 * (i + 1) + 1];
            output[2 * produced] = ToPcm16(left0 + (left1 - left0) * t);
            output[2 * produced + 1] = ToPcm16(right0 + (right1 - right0) * t);
            produced++;
            _position += _step;
        }

        _position -= frames;
        _previousLeft = _stereo[2 * (frames - 1)];
        _previousRight = _stereo[2 * (frames - 1) + 1];
        return _output.AsSpan(0, produced * AudioBuffer.BytesPerFrame);
    }

    private static short ToPcm16(float value) => (short)MathF.Round(Math.Clamp(value, -1f, 1f) * 32767f);

    private static void Grow<T>(ref T[] buffer, int length)
    {
        if (buffer.Length < length) buffer = new T[Math.Max(length, buffer.Length * 2)];
    }
}
```

```csharp
namespace ClipThy.Video;

public enum EncoderKind
{
    Nvenc,
    MediaFoundation,
}

/// <summary>The "encoder" setting: auto = NVENC with Media Foundation fallback.</summary>
public enum EncoderMode
{
    Auto,
    Nvenc,
    MediaFoundation,
}

/// <summary>Encoder choice, NVENC failure detection and restart backoff for the capture ffmpeg.</summary>
public static class EncoderPolicy
{
    private static readonly string[] NvencStartupFailureMarkers =
    [
        "Driver does not support the required nvenc API version",
        "minimum required Nvidia driver",
        "Cannot load nvEncodeAPI",
        "No capable devices found",
        "OpenEncodeSessionEx failed",
        "Could not open encoder",
        "Error while opening encoder",
    ];

    private static readonly TimeSpan[] BackoffSteps =
    [
        TimeSpan.FromSeconds(1),
        TimeSpan.FromSeconds(2),
        TimeSpan.FromSeconds(5),
        TimeSpan.FromSeconds(10),
        TimeSpan.FromSeconds(30),
    ];

    public static EncoderMode ParseMode(string setting) => setting switch
    {
        "nvenc" => EncoderMode.Nvenc,
        "mf" => EncoderMode.MediaFoundation,
        _ => EncoderMode.Auto,
    };

    public static EncoderKind Initial(EncoderMode mode) =>
        mode == EncoderMode.MediaFoundation ? EncoderKind.MediaFoundation : EncoderKind.Nvenc;

    public static bool IsNvencStartupFailure(IEnumerable<string> stderrLines) =>
        stderrLines.Any(line => line.Contains("nvenc", StringComparison.OrdinalIgnoreCase)
            && NvencStartupFailureMarkers.Any(marker => line.Contains(marker, StringComparison.OrdinalIgnoreCase)));

    /// <summary>True when a failed run means this session should switch to Media Foundation.</summary>
    public static bool ShouldFallBack(EncoderMode mode, EncoderKind current, bool producedPackets, IEnumerable<string> stderrLines) =>
        mode == EncoderMode.Auto && current == EncoderKind.Nvenc && !producedPackets && IsNvencStartupFailure(stderrLines);
```

```csharp
using System.Diagnostics;

namespace ClipThy.Export;

public sealed record FfmpegResult(int ExitCode, IReadOnlyList<string> StdErrTail);

public interface IFfmpegRunner
{
    Task<FfmpegResult> RunAsync(IReadOnlyList<string> args, CancellationToken cancellationToken = default);
}

/// <summary>Runs a short-lived ffmpeg (the export) and collects the end of its stderr.</summary>
public sealed class ProcessFfmpegRunner(Func<string> ffmpegPath, Action<Process>? onStarted = null) : IFfmpegRunner
{
    private const int TailLines = 20;

    public async Task<FfmpegResult> RunAsync(IReadOnlyList<string> args, CancellationToken cancellationToken = default)
    {
        var startInfo = new ProcessStartInfo(ffmpegPath())
        {
            UseShellExecute = false,
            CreateNoWindow = true,
            RedirectStandardError = true,
            RedirectStandardOutput = true,
        };
        foreach (string arg in args) startInfo.ArgumentList.Add(arg);

        using var process = new Process { StartInfo = startInfo };
        var tail = new Queue<string>();
        process.ErrorDataReceived += (_, e) =>
        {
            if (e.Data is null) return;
            lock (tail)
            {
                tail.Enqueue(e.Data);
                if (tail.Count > TailLines) tail.Dequeue();
            }
        };
        process.OutputDataReceived += (_, _) => { };

        process.Start();
        onStarted?.Invoke(process);
        process.BeginErrorReadLine();
        process.BeginOutputReadLine();
        await process.WaitForExitAsync(cancellationToken);

        lock (tail) return new FfmpegResult(process.ExitCode, tail.ToArray());
    }
}
```

### Task 13: Clip exporter

A press becomes a file in two stages. **Stage 1 runs for every press straight away:**
1. Record the press time.
2. Wait until the video buffer holds a frame captured after the press. This takes at least 100 ms so audio still in flight arrives, and at most 2 s.
3. Copy the video snapshot and both audio spans.

**Stage 2 is serialized:**
1. Write the temp files.
2. Run the export ffmpeg.
3. Move the MP4 into the month folder.

Because stage 1 doesn't wait in the queue, a queued save can't lose footage to buffer trimming. At most 3 saves can be pending; a 4th press is refused.

The tests use fakes, a fake clock (1 tick = 1 ms) and a fake delay that advances the clock, so they run instantly.

```csharp
using ClipThy.Audio;
using ClipThy.Config;
using ClipThy.Timing;
using ClipThy.Video;

namespace ClipThy.Export;

public enum SaveRequestOutcome
{
    Accepted,
    Busy,
}

public sealed record SaveResult(SaveMode Mode, string? Path, double LengthSeconds, string? Error)
{
    public bool Success => Error is null;
}

/// <summary>
/// Turns a hotkey press into a file. Stage 1 (per press, right away): wait until the video buffer has reached
/// the press time, then copy the video and both audio spans. Stage 2 (serialized): write temp files, run the
/// export ffmpeg, move the MP4 into the month folder.
/// </summary>
public sealed class ClipExporter
{
    public const int MaxPending = 3;
    public const string NothingToSave = "Nothing to save yet";

    private static readonly TimeSpan MinWait = TimeSpan.FromMilliseconds(100);
    private static readonly TimeSpan MaxWait = TimeSpan.FromSeconds(2);
    private static readonly TimeSpan PollInterval = TimeSpan.FromMilliseconds(20);

    private readonly IVideoSource _video;
    private readonly IAudioSource _system;
    private readonly IAudioSource _mic;
    private readonly IFfmpegRunner _ffmpeg;
    private readonly IClock _clock;
    private readonly Func<AppSettings> _settings;
    private readonly string _tempRoot;
    private readonly Func<TimeSpan, Task> _delay;
    private readonly SemaphoreSlim _exportLock = new(1, 1);
    private readonly List<Task> _running = [];
    private int _pending;

    private sealed record ClipData(VideoSnapshot Video, byte[] SystemAudio, byte[] MicAudio, double LengthSeconds);

    public ClipExporter(
        IVideoSource video,
        IAudioSource system,
        IAudioSource mic,
        IFfmpegRunner ffmpeg,
        IClock clock,
        Func<AppSettings> settings,
        string tempRoot,
        Func<TimeSpan, Task>? delay = null)
    {
        _video = video;
        _system = system;
        _mic = mic;
        _ffmpeg = ffmpeg;
        _clock = clock;
        _settings = settings;
        _tempRoot = tempRoot;
        _delay = delay ?? (d => Task.Delay(d));
    }

    /// <summary>Raised on a background thread when a save finishes (successfully or not).</summary>
    public event Action<SaveResult>? Completed;

    public int Pending => Volatile.Read(ref _pending);

    public SaveRequestOutcome RequestSave(SaveMode mode)
    {
        if (Interlocked.Increment(ref _pending) > MaxPending)
        {
            Interlocked.Decrement(ref _pending);
            return SaveRequestOutcome.Busy;
        }

        long pressQpc = _clock.QpcNow;
        DateTime pressLocal = _clock.UtcNow.ToLocalTime();
        AppSettings settings = _settings();
        Task task = Task.Run(() => SaveAsync(mode, pressQpc, pressLocal, settings));
        lock (_running)
        {
            _running.RemoveAll(t => t.IsCompleted);
            _running.Add(task);
        }
        return SaveRequestOutcome.Accepted;
    }

    /// <summary>Completes when every accepted save has finished.</summary>
    public Task WhenIdleAsync()
    {
        lock (_running) return Task.WhenAll(_running.ToArray());
    }

    /// <summary>Deletes temp folders left behind by a previous run (for example after a crash).</summary>
    public static void CleanTempRoot(string tempRoot)
    {
        if (!Directory.Exists(tempRoot)) return;
        foreach (string folder in Directory.GetDirectories(tempRoot))
        {
            try { Directory.Delete(folder, recursive: true); }
            catch (IOException) { }
            catch (UnauthorizedAccessException) { }
        }
    }

    private async Task SaveAsync(SaveMode mode, long pressQpc, DateTime pressLocal, AppSettings settings)
    {
        SaveResult result;
        try
        {
            ClipData? clip = await CaptureAsync(pressQpc, settings);
            result = clip is null
                ? new SaveResult(mode, null, 0, NothingToSave)
                : await ExportAsync(mode, clip, pressLocal, settings);
        }
        catch (Exception ex)
        {
            result = new SaveResult(mode, null, 0, ex.Message);
        }
        finally
        {
            Interlocked.Decrement(ref _pending);
        }
        Completed?.Invoke(result);
    }

    private async Task<ClipData?> CaptureAsync(long pressQpc, AppSettings settings)
    {
        long frequency = _clock.Frequency;
        long minWait = QpcTime.FromTimeSpan(MinWait, frequency);
        long maxWait = QpcTime.FromTimeSpan(MaxWait, frequency);
        long waitStart = _clock.QpcNow;
        while (true)
        {
            long waited = _clock.QpcNow - waitStart;
            bool videoCaughtUp = _video.NewestCaptureQpc is long newest && newest >= pressQpc;
            if ((waited >= minWait && videoCaughtUp) || waited >= maxWait) break;
            await _delay(PollInterval);
        }

        VideoSnapshot? video = _video.Snapshot(pressQpc, TimeSpan.FromSeconds(settings.ClipSeconds));
        if (video is null) return null;

        long audioEnd = video.LastFrameQpc + QpcTime.FromSeconds(1.0 / settings.Fps, frequency);
        return new ClipData(
            video,
            _system.Extract(video.FirstFrameQpc, audioEnd),
            _mic.Extract(video.FirstFrameQpc, audioEnd),
            QpcTime.ToSeconds(audioEnd - video.FirstFrameQpc, frequency));
    }

    private async Task<SaveResult> ExportAsync(SaveMode mode, ClipData clip, DateTime pressLocal, AppSettings settings)
    {
        await _exportLock.WaitAsync();
        string folder = Path.Combine(_tempRoot, Guid.NewGuid().ToString("N"));
        try
        {
            Directory.CreateDirectory(folder);
            string videoTs = Path.Combine(folder, "video.ts");
            string systemWav = Path.Combine(folder, "system.wav");
            string micWav = Path.Combine(folder, "mic.wav");
            string mp4 = Path.Combine(folder, "clip.mp4");

            await File.WriteAllBytesAsync(videoTs, clip.Video.Data);
            WavWriter.Write(systemWav, clip.SystemAudio);
            WavWriter.Write(micWav, clip.MicAudio);

            FfmpegResult run = await _ffmpeg.RunAsync(ExportArgs.Build(mode, videoTs, systemWav, micWav, mp4));
            if (run.ExitCode != 0 || !File.Exists(mp4))
            {
                string reason = run.StdErrTail.LastOrDefault() ?? $"exit code {run.ExitCode}";
                return new SaveResult(mode, null, 0, $"ffmpeg failed: {reason}");
            }

            string destination = OutputPaths.Reserve(settings.OutputRoot, pressLocal);
            File.Move(mp4, destination);
            return new SaveResult(mode, destination, clip.LengthSeconds, null);
        }
        finally
        {
            try { Directory.Delete(folder, recursive: true); }
            catch (IOException) { }
            catch (UnauthorizedAccessException) { }
            _exportLock.Release();
        }
    }
}
```

```csharp
    private readonly long _maxBytes;

    public FileLog(string path, long maxBytes = 1_000_000)

        _maxBytes = maxBytes;
        Directory.CreateDirectory(Path.GetDirectoryName(path)!);
    }

        Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData), "ClipThy", "clipthy.log");

    public string RolledPath => Path.Combine(
        Path.GetDirectoryName(FilePath)!,
        Path.GetFileNameWithoutExtension(FilePath) + ".1" + Path.GetExtension(FilePath));

    public void Write(string message)
    {
        string line = $"{DateTime.Now:yyyy-MM-dd HH:mm:ss.fff} {message}{Environment.NewLine}";
        lock (_lock)
        {
            try
            {
                var file = new FileInfo(FilePath);
                if (file.Exists && file.Length + line.Length > _maxBytes) File.Move(FilePath, RolledPath, overwrite: true);
                File.AppendAllText(FilePath, line);
            }
            catch (IOException) { }
            catch (UnauthorizedAccessException) { }
        }
    }
```

```csharp
using System.ComponentModel;
using System.Diagnostics;
using System.Runtime.InteropServices;

namespace ClipThy.Infrastructure;

/// <summary>
/// A Windows job object with "kill on close": processes added to it die when ClipThy exits, even on a crash.
/// The handle is deliberately never closed, because closing it kills the processes.
/// </summary>
public sealed class JobObject
{
    private const int JobObjectExtendedLimitInformation = 9;
    private const uint JOB_OBJECT_LIMIT_KILL_ON_JOB_CLOSE = 0x2000;

    private readonly IntPtr _handle;

    public JobObject()
    {
        _handle = CreateJobObject(IntPtr.Zero, null);
        if (_handle == IntPtr.Zero) throw new Win32Exception(Marshal.GetLastWin32Error());

        var info = new JOBOBJECT_EXTENDED_LIMIT_INFORMATION();
        info.BasicLimitInformation.LimitFlags = JOB_OBJECT_LIMIT_KILL_ON_JOB_CLOSE;
        if (!SetInformationJobObject(_handle, JobObjectExtendedLimitInformation, ref info, Marshal.SizeOf<JOBOBJECT_EXTENDED_LIMIT_INFORMATION>()))
        {
            throw new Win32Exception(Marshal.GetLastWin32Error());
        }
    }

    public void Add(Process process)
    {
        if (!AssignProcessToJobObject(_handle, process.Handle)) throw new Win32Exception(Marshal.GetLastWin32Error());
    }

    [DllImport("kernel32.dll", CharSet = CharSet.Unicode, SetLastError = true)]
    private static extern IntPtr CreateJobObject(IntPtr jobAttributes, string? name);

    [DllImport("kernel32.dll", SetLastError = true)]
    private static extern bool SetInformationJobObject(IntPtr job, int infoClass, ref JOBOBJECT_EXTENDED_LIMIT_INFORMATION info, int length);

    [DllImport("kernel32.dll", SetLastError = true)]
    private static extern bool AssignProcessToJobObject(IntPtr job, IntPtr process);

    [StructLayout(LayoutKind.Sequential)]
    private struct JOBOBJECT_BASIC_LIMIT_INFORMATION
    {
        public long PerProcessUserTimeLimit;
        public long PerJobUserTimeLimit;
        public uint LimitFlags;
        public UIntPtr MinimumWorkingSetSize;
        public UIntPtr MaximumWorkingSetSize;
        public uint ActiveProcessLimit;
        public UIntPtr Affinity;
        public uint PriorityClass;
        public uint SchedulingClass;
    }

    private struct IO_COUNTERS
    {
        public ulong ReadOperationCount;
        public ulong WriteOperationCount;
        public ulong OtherOperationCount;
        public ulong ReadTransferCount;
        public ulong WriteTransferCount;
        public ulong OtherTransferCount;
    }
```

### Task 15: Video capture (ffmpeg supervisor) and integration tests

**What `VideoCapture` does.** It runs the capture ffmpeg and feeds stdout to `VideoBuffer`, stamping each read with QPC and UTC. It also:
- drains stderr continuously (an undrained pipe would stall ffmpeg);
- kills ffmpeg after 3 s without output, which the watchdog detects;
- restarts ffmpeg with backoff (1 â†’ 2 â†’ 5 â†’ 10 â†’ 30 s), resetting after 60 s of healthy running;
- restarts straight away on request, with no backoff;
- stops while paused;
- switches NVENC â†’ Media Foundation when `EncoderPolicy.ShouldFallBack` says so.

An unexpected exception is logged and treated as a failed run, so a bug can't kill the capture thread and take the app down with it.

```csharp
using System.Buffers.Binary;
using System.Diagnostics;
using System.Text;
using System.Text.Json;

namespace ClipThy.Tests.TestSupport;

/// <summary>Runs ffmpeg/ffprobe from PATH for integration tests.</summary>
public static class FfmpegTool
{
    public static string Run(string exe, params string[] args)
    {
        var startInfo = new ProcessStartInfo(exe)
        {
            UseShellExecute = false,
            CreateNoWindow = true,
            RedirectStandardOutput = true,
            RedirectStandardError = true,
        };
        foreach (string arg in args) startInfo.ArgumentList.Add(arg);
        using Process process = Process.Start(startInfo)!;
        Task<string> stdout = process.StandardOutput.ReadToEndAsync();
        Task<string> stderr = process.StandardError.ReadToEndAsync();
        process.WaitForExit();
        if (process.ExitCode != 0) throw new InvalidOperationException($"{exe} exited with {process.ExitCode}: {stderr.Result}");
        return stdout.Result;
    }

    public static JsonElement Probe(string file) => JsonDocument.Parse(Run("ffprobe",
        "-v", "error",
        "-show_entries", "format=duration:stream=index,codec_type,codec_name,color_space,color_primaries,color_transfer:stream_tags=handler_name:stream_disposition=default",
        "-of", "json", file)).RootElement;

    /// <summary>Top-level MP4 box types in file order (to check that moov comes before mdat).</summary>
    public static List<string> TopLevelBoxes(string mp4)
    {
        var boxes = new List<string>();
        using FileStream file = File.OpenRead(mp4);
        var header = new byte[16];
        while (file.Position < file.Length)
        {
            file.ReadExactly(header, 0, 8);
            long size = BinaryPrimitives.ReadUInt32BigEndian(header);
            boxes.Add(Encoding.ASCII.GetString(header, 4, 4));
            if (size == 0) break; // box runs to the end of the file
            if (size == 1)
            {
                file.ReadExactly(header, 8, 8);
                size = (long)BinaryPrimitives.ReadUInt64BigEndian(header.AsSpan(8));
                file.Seek(size - 16, SeekOrigin.Current);
            }
            else
            {
                file.Seek(size - 8, SeekOrigin.Current);
            }
        }
        return boxes;
    }
}
```

```csharp
        lock (_lock)
        {
            _paused = false;
            _restartRequested = true;
        }
        _wake.Set();
    }

    public void Dispose()
    {
        lock (_lock)
        {
            _disposed = true;
            KillCurrent();
        }
        _wake.Set();
        _thread?.Join(TimeSpan.FromSeconds(5));
    }

    private void Run()
    {
        int failures = 0;
        bool notified = false;
        while (true)
        {
            _wake.Reset();
            bool paused;
            lock (_lock)
            {
                if (_disposed) return;
                paused = _paused;
                _restartRequested = false;
            }
            if (paused)
            {
                Raise(CaptureState.Paused);
                _wake.Wait();
                continue;
            }

            Raise(failures == 0 ? CaptureState.Starting : CaptureState.Restarting);
            RunResult result;
            try
            {
                result = RunOnce();
            }
            catch (Exception ex)
            {
                // A bug must not kill the capture thread (and with it the app): log it and retry with backoff.
                _log($"Video capture error: {ex}");
                lock (_lock)
                {
                    KillCurrent();
                    _process = null;
                }
                result = new RunResult(false, [ex.Message], TimeSpan.Zero);
            }

            bool intentional;
            lock (_lock)
            {
                if (_disposed) return;
                intentional = _restartRequested || _paused;
            }
            if (intentional)
            {
                failures = 0;
                continue;
            }

            if (result.Duration >= HealthyRun)
            {
                failures = 0;
                notified = false;
            }
            failures++;

            EncoderKind current = Encoder;
            if (EncoderPolicy.ShouldFallBack(_mode, current, result.ProducedPackets, result.StdErrTail))
            {
                lock (_lock) _encoder = EncoderKind.MediaFoundation;
                _log("NVENC could not start; switching to Media Foundation for this session.");
                EncoderFellBack?.Invoke(
                    "NVENC couldn't start, so ClipThy is using Windows Media Foundation on the same GPU. Updating the NVIDIA driver usually fixes this.");
                failures = 0;
                continue;
            }

            string lastLine = result.StdErrTail.LastOrDefault() ?? "ffmpeg stopped unexpectedly";
            if (failures >= NotifyAfterFailures && !notified)
            {
                notified = true;
                Raise(CaptureState.Failed, lastLine);
                PersistentFailure?.Invoke(lastLine);
            }
            else
            {
                Raise(CaptureState.Restarting, lastLine);
            }
            _wake.Wait(EncoderPolicy.Backoff(failures));
        }
    }

    private RunResult RunOnce()
    {
        EncoderKind encoder = Encoder;
        var startInfo = new ProcessStartInfo(_ffmpegPath)
        {
            UseShellExecute = false,
            CreateNoWindow = true,
            RedirectStandardOutput = true,
            RedirectStandardError = true,
        };
        foreach (string arg in _buildArgs(encoder)) startInfo.ArgumentList.Add(arg);

        var process = new Process { StartInfo = startInfo };
        var tail = new Queue<string>();
        process.ErrorDataReceived += (_, e) =>
        {
            if (e.Data is null) return;
            lock (tail)
            {
                tail.Enqueue(e.Data);
                if (tail.Count > StdErrTailLines) tail.Dequeue();
            }
        };

        long started = Stopwatch.GetTimestamp();
        try
        {
            process.Start();
        }
        catch (Win32Exception ex)
        {
            process.Dispose();
            return new RunResult(false, [$"Could not start ffmpeg ('{_ffmpegPath}'): {ex.Message}"], TimeSpan.Zero);
        }

        try { _job?.Add(process); }
        catch (Win32Exception ex) { _log($"Could not add ffmpeg to the job object: {ex.Message}"); }

        _buffer.Clear();
        lock (_lock)
        {
            _process = process;
            if (_disposed || _paused || _restartRequested) KillCurrent();
        }
        process.BeginErrorReadLine();
        _log($"Capture ffmpeg started (pid {process.Id}, {encoder}).");

        bool produced = false;
        Interlocked.Exchange(ref _lastPacketQpc, Stopwatch.GetTimestamp());
        using (new System.Threading.Timer(_ => KillIfStalled(process), null, 500, 500))
        {
            Stream stdout = process.StandardOutput.BaseStream;
            byte[] chunk = new byte[64 * 1024];
            try
            {
                int read;
                while ((read = stdout.Read(chunk, 0, chunk.Length)) > 0)
                {
                    long arrivalQpc = Stopwatch.GetTimestamp();
                    long arrivalUtc90k = MpegTs.UtcTo90k(DateTime.UtcNow);
                    _buffer.Append(chunk.AsSpan(0, read), arrivalQpc, arrivalUtc90k);
                    Interlocked.Exchange(ref _lastPacketQpc, arrivalQpc);
                    if (!produced)
                    {
                        produced = true;
                        Raise(CaptureState.Recording);
                    }
                }
            }
            catch (IOException) { }
        }

        process.WaitForExit();
        lock (_lock) _process = null;
        string[] lines;
        lock (tail) lines = tail.ToArray();
        TimeSpan duration = Stopwatch.GetElapsedTime(started);
        _log($"Capture ffmpeg exited (code {process.ExitCode}) after {duration.TotalSeconds:F1} s; timestamp warnings so far: {_buffer.TimestampWarnings}. {string.Join(" | ", lines.TakeLast(3))}");
        process.Dispose();
        return new RunResult(produced, lines, duration);
    }

    private void KillIfStalled(Process process)
    {
        long silentFor = Stopwatch.GetTimestamp() - Interlocked.Read(ref _lastPacketQpc);
        if (silentFor < QpcTime.FromTimeSpan(StallTimeout, Stopwatch.Frequency)) return;
        _log("No video for 3 s; restarting ffmpeg.");
        Interlocked.Exchange(ref _lastPacketQpc, Stopwatch.GetTimestamp());
        try { process.Kill(entireProcessTree: true); }
        catch (InvalidOperationException) { }
        catch (Win32Exception) { }
    }

    private void KillCurrent()
    {
        try { _process?.Kill(entireProcessTree: true); }
        catch (InvalidOperationException) { }
        catch (Win32Exception) { }
    }

    private void Raise(CaptureState state, string? detail = null) =>
        StatusChanged?.Invoke(new CaptureStatus(state, Encoder, detail));
}
```

```csharp
using System.Diagnostics;
using System.Runtime.InteropServices;
using ClipThy.Timing;
using NAudio.CoreAudioApi;
using NAudio.CoreAudioApi.Interfaces;
using NAudio.Wave;

namespace ClipThy.Audio;

public enum AudioSourceKind
{
    SystemLoopback,
    Microphone,
}

/// <summary>
/// Captures the default render device (loopback) or default microphone with WASAPI in shared mode, stamps
/// every packet with WASAPI's own capture time, and appends it to an <see cref="AudioBuffer"/>.
/// Follows default-device changes and survives unplugging.
/// </summary>
public sealed class AudioCapture : IDisposable
{
    private static readonly Guid SubtypeIeeeFloat = new("00000003-0000-0010-8000-00aa00389b71");
    private static readonly Guid SubtypePcm = new("00000001-0000-0010-8000-00aa00389b71");
    private const int ENotFound = unchecked((int)0x80070490);
    private const int EAccessDenied = unchecked((int)0x80070005);
    private const long BufferDuration100ns = 1_000_000; // 100 ms

    private readonly AudioSourceKind _kind;
    private readonly AudioBuffer _buffer;
    private readonly Action<string> _log;
    private readonly AutoResetEvent _deviceChanged = new(false);
    private readonly ManualResetEvent _stop = new(false);
    private Thread? _thread;
    private string? _lastProblem;

    public AudioCapture(AudioSourceKind kind, AudioBuffer buffer, Action<string> log)
    {
        _kind = kind;
        _buffer = buffer;
        _log = log;
    }

    /// <summary>Raised once per distinct problem (no device, access blocked), on the capture thread.</summary>
    public event Action<string>? Problem;

    private DataFlow Flow => _kind == AudioSourceKind.SystemLoopback ? DataFlow.Render : DataFlow.Capture;

    public void Start()
    {
        _thread = new Thread(Run) { IsBackground = true, Name = $"ClipThy {_kind} capture" };
        _thread.SetApartmentState(ApartmentState.MTA);
        _thread.Start();
    }

    /// <summary>Reopens the default device (for example after resume from sleep).</summary>
    public void Restart() => _deviceChanged.Set();

    public void Dispose()
    {
        _stop.Set();
        _thread?.Join(TimeSpan.FromSeconds(3));
    }

    private void Run()
    {
        using var enumerator = new MMDeviceEnumerator();
        var notifications = new DefaultDeviceNotifications(this);
        enumerator.RegisterEndpointNotificationCallback(notifications);
        try
        {
            while (!_stop.WaitOne(0))
            {
                try
                {
                    CaptureUntilDeviceChanges(enumerator);
                }
                catch (COMException ex) when (ex.HResult == ENotFound)
                {
                    ReportProblem(_kind == AudioSourceKind.Microphone
                        ? "No microphone found; the mic track will be silent."
                        : "No audio output device found; the system audio track will be silent.");
                    WaitHandle.WaitAny([_deviceChanged, _stop], TimeSpan.FromSeconds(5));
                }
                catch (Exception ex) when (ex.HResult == EAccessDenied || ex is UnauthorizedAccessException)
                {
                    ReportProblem("Windows is blocking microphone access (Settings > Privacy > Microphone); the mic track will be silent.");
                    WaitHandle.WaitAny([_deviceChanged, _stop], TimeSpan.FromSeconds(5));
                }
                catch (Exception ex)
                {
                    // Device unplugged (AUDCLNT_E_DEVICE_INVALIDATED) or anything unexpected: log and reopen.
                    _log($"{_kind} capture error: {ex.Message}. Reopening.");
                    _stop.WaitOne(TimeSpan.FromSeconds(1));
                }
            }
        }
        finally
        {
            enumerator.UnregisterEndpointNotificationCallback(notifications);
        }
    }

    private void CaptureUntilDeviceChanges(MMDeviceEnumerator enumerator)
    {
        using MMDevice device = enumerator.GetDefaultAudioEndpoint(Flow, Role.Console);
        AudioClient client = device.AudioClient;
        try
        {
            WaveFormat mixFormat = client.MixFormat;
            InputFormat format = ToInputFormat(mixFormat);
            var converter = new PcmConverter(format);
            using var packetReady = new AutoResetEvent(false);

            AudioClientStreamFlags flags = AudioClientStreamFlags.EventCallback;
            if (_kind == AudioSourceKind.SystemLoopback) flags |= AudioClientStreamFlags.Loopback;
            client.Initialize(AudioClientShareMode.Shared, flags, BufferDuration100ns, 0, mixFormat, Guid.Empty);
            client.SetEventHandle(packetReady.SafeWaitHandle.DangerousGetHandle());
            AudioCaptureClient capture = client.AudioCaptureClient;

            byte[] raw = new byte[format.BytesPerFrame * 4800];
            client.Start();
            _log($"{_kind} capture started on '{device.FriendlyName}' ({mixFormat}).");
            _lastProblem = null;

            WaitHandle[] handles = [packetReady, _deviceChanged, _stop];
            while (true)
            {
                // The 10 ms timeout also covers systems where loopback never signals the event.
                int signaled = WaitHandle.WaitAny(handles, 10);
                if (signaled is 1 or 2)
                {
                    _log($"{_kind} capture: default device changed or stopping.");
                    return;
                }

                while (capture.GetNextPacketSize() > 0)
                {
                    IntPtr data = capture.GetBuffer(out int frames, out AudioClientBufferFlags bufferFlags, out _, out long qpcPosition);
                    int bytes = frames * format.BytesPerFrame;
                    if (raw.Length < bytes) raw = new byte[bytes];
                    if ((bufferFlags & AudioClientBufferFlags.Silent) != 0) Array.Clear(raw, 0, bytes);
                    else Marshal.Copy(data, raw, 0, bytes);
                    capture.ReleaseBuffer(frames);

                    long startQpc = (bufferFlags & AudioClientBufferFlags.TimestampError) != 0
                        ? Stopwatch.GetTimestamp() - QpcTime.FromFrames(frames, format.SampleRate, Stopwatch.Frequency)
                        : QpcTime.From100ns(qpcPosition, Stopwatch.Frequency);
                    _buffer.Append(converter.Convert(raw.AsSpan(0, bytes)), startQpc);
                }
            }
        }
        finally
        {
            try { client.Stop(); } catch (COMException) { }
            client.Dispose();
        }
    }

    private static InputFormat ToInputFormat(WaveFormat format)
    {
        bool isFloat = format.Encoding == WaveFormatEncoding.IeeeFloat
            || (format is WaveFormatExtensible ext && ext.SubFormat == SubtypeIeeeFloat);
        bool isPcm = format.Encoding == WaveFormatEncoding.Pcm
            || (format is WaveFormatExtensible pcm && pcm.SubFormat == SubtypePcm);

        if (isFloat && format.BitsPerSample == 32) return new InputFormat(format.SampleRate, format.Channels, SampleFormat.Float32);
        if (isPcm && format.BitsPerSample == 16) return new InputFormat(format.SampleRate, format.Channels, SampleFormat.Pcm16);
        throw new NotSupportedException($"Unsupported mix format: {format}");
    }

    private void ReportProblem(string message)
    {
        if (message == _lastProblem) return;
        _lastProblem = message;
        _log(message);
        Problem?.Invoke(message);
    }

    private sealed class DefaultDeviceNotifications(AudioCapture owner) : IMMNotificationClient
    {
        public void OnDefaultDeviceChanged(DataFlow flow, Role role, string defaultDeviceId)
        {
            if (flow == owner.Flow && role == Role.Console) owner._deviceChanged.Set();
        }

        public void OnDeviceAdded(string pwstrDeviceId) { }

        public void OnDeviceRemoved(string deviceId) { }

        public void OnDeviceStateChanged(string deviceId, DeviceState newState) { }

        public void OnPropertyValueChanged(string pwstrDeviceId, PropertyKey key) { }
    }
}
```

```csharp
using System.Runtime.InteropServices;
using ClipThy.Export;

namespace ClipThy.Hotkeys;

/// <summary>Registers global hotkeys on a hidden message-only window and raises SaveRequested when pressed.</summary>
public sealed class HotkeyListener : NativeWindow, IDisposable
{
    private const int WM_HOTKEY = 0x0312;
    private const uint MOD_NOREPEAT = 0x4000;
    private static readonly IntPtr HWND_MESSAGE = new(-3);

    private readonly Dictionary<int, SaveMode> _registered = [];
    private int _nextId = 1;

    public HotkeyListener() => CreateHandle(new CreateParams { Parent = HWND_MESSAGE });

    public event Action<SaveMode>? SaveRequested;

    /// <summary>Returns false when another program already owns the hotkey.</summary>
    public bool Register(Hotkey hotkey, SaveMode mode)
    {
        int id = _nextId++;
        if (!RegisterHotKey(Handle, id, (uint)hotkey.Modifiers | MOD_NOREPEAT, (uint)hotkey.Key)) return false;
        _registered[id] = mode;
        return true;
    }

    protected override void WndProc(ref Message m)
    {
        if (m.Msg == WM_HOTKEY && _registered.TryGetValue((int)m.WParam, out SaveMode mode)) SaveRequested?.Invoke(mode);
        base.WndProc(ref m);
    }

    public void Dispose()
    {
        foreach (int id in _registered.Keys) UnregisterHotKey(Handle, id);
        _registered.Clear();
        DestroyHandle();
    }

    [DllImport("user32.dll", SetLastError = true)]
    private static extern bool RegisterHotKey(IntPtr hWnd, int id, uint modifiers, uint virtualKey);

    [DllImport("user32.dll", SetLastError = true)]
    private static extern bool UnregisterHotKey(IntPtr hWnd, int id);
}
```

```csharp
    public static SoundPlayer SaveSound()
    {
        const int rate = 48_000;
        int frames = rate * 80 / 1000;
        var pcm = new byte[frames * 2];
        for (int i = 0; i < frames; i++)
        {
            double t = (double)i / rate;
            double frequency = i < frames / 2 ? 880 : 1320;
            double fade = Math.Min(1.0, Math.Min(i, frames - i) / (rate * 0.005));

            BitConverter.TryWriteBytes(pcm.AsSpan(i * 2), sample);
        }
        var stream = new MemoryStream();
        WavWriter.Write(stream, pcm, rate, channels: 1, bitsPerSample: 16);
        stream.Position = 0;
        var player = new SoundPlayer(stream);
        player.Load();
```

```csharp
        _micCapture.Problem += message => _ui.Post(_ => ShowBalloon("ClipThy audio", message, ToolTipIcon.Warning), null);

        int fps = _settings.Fps;
        int bitrate = _settings.VideoBitrateMbps;
        _capture = new VideoCapture(
            _video,
            encoder => CaptureArgs.Build(encoder, fps, bitrate, CaptureArgs.DesktopGraph(fps)),
            _settings.FfmpegPath,
            mode,
            _job,
            _log.Write);
        _status = new CaptureStatus(CaptureState.Starting, EncoderPolicy.Initial(mode));
        _capture.StatusChanged += status => _ui.Post(_ => UpdateStatus(status), null);
        _capture.EncoderFellBack += message => _ui.Post(_ => ShowBalloon("ClipThy", message, ToolTipIcon.Info), null);
        _capture.PersistentFailure += line => _ui.Post(_ => ShowBalloon("ClipThy can't capture the screen", line, ToolTipIcon.Error), null);

            _video,
            _systemAudio,
            _micAudio,

            SystemClock.Instance,
            () => _settings,

        SystemEvents.DisplaySettingsChanged += OnDisplaySettingsChanged;
        SystemEvents.TimeChanged += OnTimeChanged;
        SystemEvents.SessionSwitch += OnSessionSwitch;

        _systemCapture.Start();
        _micCapture.Start();
        _capture.Start();

    {

            ? AppSettings.ClipLengthPresets

            ShowBalloon("ClipThy", "Still saving – try again in a moment.", ToolTipIcon.Info);

    private void OnSaved(SaveResult result)
    {
        if (!result.Success)
        {
            _log.Write($"Save failed: {result.Error}");
            ShowBalloon("ClipThy couldn't save the clip", result.Error!, ToolTipIcon.Error);
            return;
        }
        _log.Write($"Saved {result.Path} ({result.LengthSeconds:F1} s, {result.Mode}).");
        if (_settings.PlaySound) _saveSound.Play();
        ShowBalloon($"Clip saved – {Math.Round(result.LengthSeconds)} s", Path.GetFileName(result.Path!), ToolTipIcon.Info, result.Path);
    }

    private void OnExportStarted(Process process)
    {
        try
        {
            process.PriorityClass = ProcessPriorityClass.BelowNormal;
            _job.Add(process);
        }
        catch (Exception ex) when (ex is InvalidOperationException or System.ComponentModel.Win32Exception)
        {
            _log.Write($"Could not adjust export ffmpeg: {ex.Message}");
        }
    }

    private void SetClipLength(int seconds)
    {
        _settings = _settings with { ClipSeconds = seconds };
        _store.Save(_settings);
        ApplyRetention();
        RebuildLengthMenu();
        UpdateStatus(_status);
    }

    private void ApplyRetention()
    {
        _video.Retention = TimeSpan.FromSeconds(_settings.ClipSeconds + 2);
        _systemAudio.Retention = TimeSpan.FromSeconds(_settings.ClipSeconds + 3);
        _micAudio.Retention = TimeSpan.FromSeconds(_settings.ClipSeconds + 3);
    }

    private void ToggleAutostart()
    {
        if (_autostart.IsEnabled) _autostart.Disable();
        else _autostart.Enable(Environment.ProcessPath!);
        _autostartItem.Checked = _autostart.IsEnabled;
    }

    private void OpenClipsFolder()
    {
        string folder = OutputPaths.MonthFolder(_settings.OutputRoot, DateTime.Now);
        if (!Directory.Exists(folder)) folder = _settings.OutputRoot;
        Directory.CreateDirectory(folder);
        Process.Start("explorer.exe", $"\"{folder}\"");
    }

    private void RevealBalloonFile()
    {
        if (_balloonPath is not null && File.Exists(_balloonPath))
        {
            Process.Start("explorer.exe", $"/select,\"{_balloonPath}\"");
        }
    }

    private void UpdateStatus(CaptureStatus status)
    {
        _status = status;
        string encoder = status.Encoder == EncoderKind.Nvenc ? "NVENC" : "Media Foundation";
        string text = status.State switch
        {
            CaptureState.Recording => $"ClipThy – recording ({_settings.ClipSeconds} s, {encoder})",
            CaptureState.Paused => "ClipThy – paused (locked)",
            CaptureState.Failed => $"ClipThy – error: {status.Detail}",
            _ => "ClipThy – restarting capture…",
        };
        _icon.Text = text.Length > 127 ? text[..127] : text;
        _icon.Icon = status.State == CaptureState.Recording ? _recordingIcon : _idleIcon;
    }

    private void ShowBalloon(string title, string text, ToolTipIcon icon, string? path = null)
    {
        _balloonPath = path;
        _icon.ShowBalloonTip(5000, title, text, icon);
    }

    private void OnPowerModeChanged(object? sender, PowerModeChangedEventArgs e)
    {
        if (e.Mode != PowerModes.Resume) return;
        _capture.RequestRestart("resumed from sleep");
        _systemCapture.Restart();
        _micCapture.Restart();
    }

    private void OnDisplaySettingsChanged(object? sender, EventArgs e) => _capture.RequestRestart("display settings changed");

    private void OnTimeChanged(object? sender, EventArgs e) => _capture.RequestRestart("system clock changed");

    private void OnSessionSwitch(object? sender, SessionSwitchEventArgs e)
    {
        if (e.Reason == SessionSwitchReason.SessionLock) _capture.Pause();
        else if (e.Reason == SessionSwitchReason.SessionUnlock) _capture.Resume();
    }

    private void Exit()
    {
        _log.Write("ClipThy exiting.");
        SystemEvents.PowerModeChanged -= OnPowerModeChanged;
        SystemEvents.DisplaySettingsChanged -= OnDisplaySettingsChanged;
        SystemEvents.TimeChanged -= OnTimeChanged;
        SystemEvents.SessionSwitch -= OnSessionSwitch;
        _hotkeys.Dispose();
        _capture.Dispose();
        _systemCapture.Dispose();
        _micCapture.Dispose();
        _exporter.WhenIdleAsync().Wait(TimeSpan.FromSeconds(10));
        _icon.Visible = false;
        _icon.Dispose();
        ExitThread();
    }
}
```

```csharp
using NAudio.CoreAudioApi;
using NAudio.Wave;

namespace ClipThy.Tray;

/// <summary>
/// "ClipThy.exe --sync-test": an 800×800 window on the primary monitor that flashes white and beeps at the
/// same moment once per second. Record a clip while it runs and analyse it with tools\sync-check.ps1.
/// </summary>
public sealed class SyncTestForm : Form
{
    private readonly WasapiOut _output = new(AudioClientShareMode.Shared, useEventSync: true, latency: 20);

    public SyncTestForm()
    {
        Text = "ClipThy sync test – close this window to stop";
        BackColor = Color.Black;
        StartPosition = FormStartPosition.Manual;
        Rectangle screen = Screen.PrimaryScreen!.Bounds;
        Bounds = new Rectangle(screen.X + (screen.Width - 800) / 2, screen.Y + (screen.Height - 800) / 2, 800, 800);
        TopMost = true;

        var beeps = new BeepProvider();
        beeps.BeepStarted += () => BeginInvoke(Flash);
        _output.Init(beeps);
        Shown += (_, _) => _output.Play();
        FormClosed += (_, _) => _output.Dispose();
    }

    private async void Flash()
    {
        BackColor = Color.White;
        await Task.Delay(100);
        BackColor = Color.Black;
    }

    /// <summary>48 kHz stereo silence with a 50 ms 1 kHz beep at the start of every second.</summary>
    private sealed class BeepProvider : ISampleProvider
    {
        private const int Rate = 48_000;
        private long _frame;

        public WaveFormat WaveFormat { get; } = WaveFormat.CreateIeeeFloatWaveFormat(Rate, 2);

        public event Action? BeepStarted;

        public int Read(float[] buffer, int offset, int count)
        {
            for (int i = 0; i < count / 2; i++, _frame++)
            {
                long n = _frame % Rate;
                if (n == 0) BeepStarted?.Invoke();
                float sample = n < Rate / 20 ? 0.5f * MathF.Sin(2 * MathF.PI * 1000 * n / Rate) : 0f;
                buffer[offset + 2 * i] = sample;
                buffer[offset + 2 * i + 1] = sample;
            }
            return count;
        }
    }
}
```

```powershell
<#
.SYNOPSIS
Measures A/V offset and dropped frames in a clip recorded while "ClipThy.exe --sync-test" was running.

.DESCRIPTION
The sync test window flashes white and beeps at the same moment once per second. This script finds the
flashes (jumps in average brightness) and the beeps (ends of silence on the system audio track), pairs
them, and reports audio-minus-video offset. It also counts dropped frames from gaps in frame timestamps.
Positive offset = audio late. The test window itself adds roughly 10-20 ms (its beep is generated a
buffer ahead of the flash being drawn), so read results with that in mind.

.EXAMPLE
powershell -ExecutionPolicy Bypass -File tools\sync-check.ps1 -Clip "C:\Users\BLOXY\Documents\ShareX\Screenshots\2026-10\clip_2026-10-01_12-00-00.mp4"
#>
param(
    [Parameter(Mandatory = $true)][string]$Clip,
    [double]$FlashThreshold = 15,
    [string]$Ffmpeg = "ffmpeg",
    [string]$Ffprobe = "ffprobe"
)

$ErrorActionPreference = "Stop"
$clipPath = (Resolve-Path $Clip).Path
$work = Join-Path ([IO.Path]::GetTempPath()) ("clipthy-sync-" + [Guid]::NewGuid().ToString("N"))
New-Item -ItemType Directory $work | Out-Null
Push-Location $work
try {
    # Frame times and average brightness. Relative file names avoid escaping "C:" inside filter arguments.
    & $Ffmpeg -hide_banner -nostdin -v error -i $clipPath -map 0:v:0 -vf "signalstats,metadata=print:key=lavfi.signalstats.YAVG:file=yavg.txt" -f null NUL
    $frames = New-Object System.Collections.Generic.List[object]
    $time = $null
    foreach ($line in Get-Content yavg.txt) {
        if ($line -match "pts_time:([0-9.]+)") { $time = [double]$Matches[1] }
        elseif ($line -match "YAVG=([0-9.]+)") { $frames.Add([pscustomobject]@{ Time = $time; Y = [double]$Matches[1] }) }
    }
    $flashes = @()
    for ($i = 1; $i -lt $frames.Count; $i++) {
        if ($frames[$i].Y - $frames[$i - 1].Y -gt $FlashThreshold) { $flashes += $frames[$i].Time }
    }

    # Beep onsets: the System track of a multitrack (Alt+F11) clip, otherwise the only track.
    $audioTracks = @(& $Ffprobe -v error -select_streams a -show_entries stream=index -of csv=p=0 $clipPath).Count
    $track = 0
    if ($audioTracks -ge 3) { $track = 1 }
    & $Ffmpeg -hide_banner -nostdin -v error -i $clipPath -map "0:a:$track" -af "silencedetect=noise=-30dB:d=0.3,ametadata=print:key=lavfi.silence_end:file=silence.txt" -f null NUL
    $beeps = @()
    foreach ($line in Get-Content silence.txt) {
        if ($line -match "lavfi.silence_end=([0-9.]+)") { $beeps += [double]$Matches[1] }
    }

    # Pair each flash with the nearest beep within half a second.
    $offsets = @()
    foreach ($flash in $flashes) {
        $nearest = $null
        foreach ($beep in $beeps) {
            if ($null -eq $nearest -or [Math]::Abs($beep - $flash) -lt [Math]::Abs($nearest - $flash)) { $nearest = $beep }
        }
        if ($null -ne $nearest -and [Math]::Abs($nearest - $flash) -le 0.5) { $offsets += ($nearest - $flash) * 1000 }
    }

    # Dropped frames: gaps longer than 1.5x the typical frame interval.
    $intervals = @()
    for ($i = 1; $i -lt $frames.Count; $i++) { $intervals += $frames[$i].Time - $frames[$i - 1].Time }
    $sorted = $intervals | Sort-Object
    $median = $sorted[[int]($sorted.Count / 2)]
    $dropped = 0
    foreach ($interval in $intervals) {
        if ($interval -gt 1.5 * $median) { $dropped += [Math]::Round($interval / $median) - 1 }
    }
    $droppedPercent = 100.0 * $dropped / ($frames.Count + $dropped)

    Write-Host ("Dropped frames: {0} of {1} ({2:F2} %)" -f $dropped, ($frames.Count + $dropped), $droppedPercent)
    Write-Host ("Flashes: {0}, beeps: {1}, matched: {2}" -f $flashes.Count, $beeps.Count, $offsets.Count)
    if ($offsets.Count -eq 0) {
        Write-Host "No flash/beep pairs found, so no A/V offset (expected for a normal clip; for a sync test, was the window on the primary monitor with sound on?)"
        exit 2
    }
    $stats = $offsets | Measure-Object -Average -Minimum -Maximum
    Write-Host ("A/V offset (audio minus video): mean {0:F1} ms, min {1:F1} ms, max {2:F1} ms (positive = audio late)" -f $stats.Average, $stats.Minimum, $stats.Maximum)

    $pass = [Math]::Abs($stats.Average) -le 40 -and $droppedPercent -le 1
    if ($pass) { Write-Host "PASS (|mean offset| <= 40 ms, dropped frames <= 1 %)" } else { Write-Host "FAIL (targets: |mean offset| <= 40 ms, dropped frames <= 1 %)" }
    if ($pass) { exit 0 } else { exit 1 }
}
finally {
    Pop-Location
    Remove-Item -Recurse -Force $work
}
```
